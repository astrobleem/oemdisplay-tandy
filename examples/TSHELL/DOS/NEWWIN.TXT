%TSDRV%%TSWIN%\WIN.COM /R
IF ERRORLEVEL 1 GOTO WINFAIL
REM Show the night screen only after successful Windows return and recovery.
%TSHOME%\SHELLSEL.EXE FINISH %TSDRV%%TSWIN%\SYSTEM.INI
IF ERRORLEVEL 1 GOTO RECFAIL
GOTO DONE
:WINFAIL
ECHO Windows returned an error. Attempting shell recovery.
REM Do not skip recovery merely because WIN returned an error.
%TSHOME%\SHELLSEL.EXE RECOVER %TSDRV%%TSWIN%\SYSTEM.INI
IF ERRORLEVEL 1 GOTO RECFAIL
GOTO DONE
